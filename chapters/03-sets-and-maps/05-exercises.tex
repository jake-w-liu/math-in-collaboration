\section*{Exercises}
\begin{exercise}\label{ex:3.1}For $A=\{1,2,4\}$ and $B=\{2,3,4\}$, find $A\cup B$, $A\cap B$, and $A\setminus B$.\end{exercise}
\begin{exercise}\label{ex:3.2}Let $f:A\to B$ and $g:B\to C$ be surjective. Prove that $g\circ f:A\to C$ is surjective. Think about which witness you must find first.\end{exercise}
\begin{exercise}\label{ex:3.3}Let $f:\ZZ\to\ZZ$ be given by $f(n)=2n$. Decide whether $f$ is injective and whether it is surjective, and prove your answers. Then keep the domain and the rule, change only the codomain, and obtain a bijection.\end{exercise}
\begin{exercise}\label{ex:3.4}Let $f:A\to B$ and $S,T\subseteq B$. Prove that $f^{-1}(S\cup T)=f^{-1}(S)\cup f^{-1}(T)$ by following the membership conditions on both sides.\end{exercise}
\begin{exercise}\label{ex:3.5}Let $f:A\to B$ and $h:B\to A$. Suppose that $h\circ f=\id_A$; such an $h$ is called a \emph{left inverse} of $f$. Prove that $f$ is injective. Suppose instead that $f\circ h=\id_B$; such an $h$ is called a \emph{right inverse} of $f$. Prove that $f$ is surjective.\end{exercise}
\begin{exercise}\label{ex:3.6}Let $f:A\to B$ and $S,T\subseteq A$. Is $f(S\cap T)=f(S)\cap f(T)$ always true? Prove the containment that always holds, and give a counterexample to equality.\end{exercise}
