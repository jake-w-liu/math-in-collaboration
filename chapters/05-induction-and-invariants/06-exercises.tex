\section*{Exercises}
\begin{exercise}\label{ex:5.1}Prove by induction that $\sum_{j=1}^{n}j=n(n+1)/2$ for every $n\in\NN$. Write out every line of algebra in the induction step.\end{exercise}
\begin{exercise}\label{ex:5.2}Let $y_0=3$ and $y_{n+1}=2y_n$ for every $n\in\NN$. Compute $y_1$, $y_2$ and $y_3$, guess a formula for $y_n$, and prove your formula by induction.\end{exercise}
\begin{exercise}\label{ex:5.3}Run Euclid's algorithm on the pair $(1071,462)$. Record each quotient and remainder, the successive pairs, and the greatest common divisor that the algorithm outputs.\end{exercise}
\begin{exercise}\label{ex:5.4}A row of five signs starts with all five signs plus. As in Section~\ref{ch05:sec:invariant}, a legal move flips the signs in two different positions. Can some sequence of legal moves produce a row with exactly three minus signs? Justify your answer without listing sequences of moves.\end{exercise}
\begin{exercise}\label{ex:5.5}For the recurrence $x_{n+1}=(x_n+1)/2$ with an arbitrary real starting value $x_0$, prove the error formula $x_n-1=2^{-n}(x_0-1)$ for every $n\in\NN$ by induction. Write out both the base case and the induction step in full. Then explain why the error $x_n-1$ is never exactly zero when $x_0\ne1$.\end{exercise}
\begin{exercise}\label{ex:5.6}A proof establishes $P(0)$, and it shows that for every natural number $k$, if $P(k)$ is true, then $P(k+2)$ is true. For which natural numbers $n$ does it follow that $P(n)$ is true? Which additional base case would give $P(n)$ for every natural number $n$? Explain why.\end{exercise}
